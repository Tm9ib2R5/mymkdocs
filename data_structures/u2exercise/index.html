编写实现对顺序表的逆置

void invert_seq(SeqList *L)
{
    if (!L || L->n < 2) return;
    for (int i = 0; i < L->n / 2; ++i)   // 整除
    {
        ElemType tmp       = L->elem[i];
        L->elem[i]         = L->elem[L->n - 1 - i];
        L->elem[L->n - 1 - i] = tmp;
    }
}

编写程序实现将数据域中最小元素放置在单链表最前面

Status swap(SingleList *L)          /* 假设 Status 就是 int */
{
    Node *p, *pFront, *min, *minFront;
    if (!L || !L->first) return 0;  /* 空表 */

    pFront = NULL;
    p      = L->first->link;
    min      = L->first;
    minFront = NULL;

    /* 1. 找最小结点 */
    while (p) {
        if (p->element < min->element) {
            min      = p;
            minFront = pFront;
        }
        pFront = p;
        p      = p->link;
    }

    /* 2. 如果最小结点已在头部，直接返回 */
    if (min == L->first) return 1;

    /* 3. 把 min 从链中摘下 */
    minFront->link = min->link;

    /* 4. 头插 */
    min->link  = L->first;
    L->first   = min;
    return 1;
}

编写程序实现对单链表的逆置

void invert_list(SingleList *L)
{
    if (!L || !L->first) return;
    Node *p = L->first;   // 旧链头
    L->first = NULL;      // 新链头

    while (p) {
        Node *next = p->link; // 保存后继
        p->link = L->first;   // 头插
        L->first = p;
        p = next;
    }
}

编写程序将有序递增的单链表中数值在a到b(a<=b)增加的元素删除

int delete_range(SingleList *L, ElemType a, ElemType b)
{
    if (!L || !L->first || a > b) return 0;

    Node *prev = NULL, *p = L->first;
    while (p && p->element <= b) {
        if (p->element >= a) {          // 命中，删除
            Node *tmp = p;
            if (!prev) {                // 删头结点
                L->first = p->link;
                p = L->first;
            } else {
                prev->link = p->link;
                p = prev->link;
            }
            free(tmp);
            --L->n;
        } else {                        // 继续扫描
            prev = p;
            p    = p->link;
        }
    }
    return 1;
}

编写程序删除单链表中所有关键字值为x的元素

int delete_all_x(SingleList *L, ElemType x)
{
    if (!L || !L->first) return 0;

    Node *prev = NULL, *p = L->first;
    while (p) {
        if (p->element == x) {
            Node *tmp = p;
            if (!prev) {                // 删头
                L->first = p->link;
                p = L->first;
            } else {
                prev->link = p->link;
                p = prev->link;
            }
            free(tmp);
            --L->n;
        } else {
            prev = p;
            p    = p->link;
        }
    }
    return 1;
}
